实验 1.1：最小可用的 ReAct Agent

目标：不依赖任何框架，用 100 行 Python 代码实现 ReAct 循环。

minimal_react.py 
"""
最小 ReAct Agent：100 行代码理解 Agent 本质
依赖: pip install openai python-dotenv
"""
import os, re, json
from openai import OpenAI

client = OpenAI(api_key=os.environ["OPENAI_API_KEY"])

# === 工具定义 ===
def get_weather(city: str) -> str:
    # 实际应调用天气 API
    return f"{city} 当前 22°C 晴朗"

def search(query: str) -> str:
    # 实际应调用搜索 API
    return f"关于 '{query}' 的搜索结果..."

TOOLS = {"get_weather": get_weather, "search": search}

TOOL_DESC = """
可用工具:
- get_weather(city): 查询某城市天气
- search(query): 网络搜索
- Finish(answer): 输出最终答案
"""

REACT_PROMPT = """你是一个智能助手,可以调用以下工具:
{tools}

历史交互:
{history}

请按以下格式回应:
思考: [你对当前问题的分析]
行动: [工具名(参数)]
观察: [工具返回的结果,会自动填充]
...
当任务完成时: Finish([最终答案])

用户问题: {question}
"""

def react_agent(question: str, max_steps: int = 8) -> str:
    history = ""
    prompt = REACT_PROMPT.format(tools=TOOL_DESC, history=history, question=question)

    for step in range(max_steps):
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}]
        )
        text = response.choices[0].message.content

        # 解析行动
        action_match = re.search(r"行动:\s*(\w+)\((.*?)\)", text)
        if not action_match: break

        tool_name = action_match.group(1)
        args_str = action_match.group(2)

        if tool_name == "Finish":
            return args_str.strip('"\'')

        # 执行工具
        if tool_name in TOOLS:
            observation = TOOLS[tool_name](args_str.strip('"\''))
        else:
            observation = f"未知工具: {tool_name}"

        history += text + f"\n观察: {observation}\n"
        prompt = REACT_PROMPT.format(tools=TOOL_DESC, history=history, question=question)

    return "未能完成"

# 运行
if __name__ == "__main__":
    answer = react_agent("北京今天天气如何？适合穿什么衣服?")
    print(f"答案: {answer}")

实验 1.2：Function Calling 的原生流程

目标：用 OpenAI 原生 Function Calling 实现带工具的对话 Agent，理解模型如何决策工具调用。

function_calling_demo.py 
from openai import OpenAI
import json

client = OpenAI()

# 工具定义: JSON Schema 格式
tools = [{
    "type": "function",
    "function": {
        "name": "get_current_weather",
        "description": "获取指定城市的当前天气",
        "parameters": {
            "type": "object",
            "properties": {
                "city": {"type": "string", "description": "城市名,如 '北京'"},
                "unit": {"type": "string", "enum": ["celsius", "fahrenheit"]}
            },
            "required": ["city"]
        }
    }
}]

def run_conversation(user_msg: str):
    messages = [{"role": "user", "content": user_msg}]

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=messages,
        tools=tools,
        tool_choice="auto"  # 让模型自主决定
    )

    response_message = response.choices[0].message
    tool_calls = response_message.tool_calls

    if tool_calls:
        messages.append(response_message)
        for tool_call in tool_calls:
            args = json.loads(tool_call.function.arguments)
            # 模拟工具执行
            result = {"temperature": 22, "condition": "晴朗"}
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": json.dumps(result, ensure_ascii=False)
            })

        second_response = client.chat.completions.create(
            model="gpt-4o-mini", messages=messages
        )
        return second_response.choices[0].message.content
    return response_message.content

print(run_conversation("上海现在天气怎么样?"))

实验 3.1：长期记忆系统

目标：为一个对话 Agent 加入"它能记住用户偏好"的能力，使用向量数据库 + 摘要策略。

memory_system.py 
# 长期记忆系统的三种实现方式对比
from langchain.memory import (
    ConversationBufferMemory,      # 全量缓冲
    ConversationSummaryMemory,    # 摘要压缩
    ConversationSummaryBufferMemory, # 最近全量 + 远期摘要
    VectorStoreRetrieverMemory    # 向量检索
)

# 实战: 组合使用
class HybridMemory:
    def __init__(self, llm, vector_store, max_tokens=2000):
        self.short_term = ConversationBufferMemory(
            max_token_limit=max_tokens, return_messages=True
        )
        self.summary = ConversationSummaryMemory(llm=llm)
        self.long_term = VectorStoreRetrieverMemory(retriever=vector_store.as_retriever(k=3))

    def load(self, query):
        recent = self.short_term.load_memory_variables({})["history"]
        relevant = self.long_term.load_memory_variables({"input": query})["history"]
        return recent + relevant

实验 3.2：Plan-and-Execute 规划器

目标：让 Agent 先制定计划，再分步执行，并在失败时重规划。

planner.py 
from langchain import hub
from langgraph import StateGraph, END
from typing import TypedDict, List

class PlanExecuteState(TypedDict):
    input: str
    plan: List[str]
    past_steps: List[tuple]
    response: str

def planner_node(state):
    # 用 LLM 生成步骤
    plan = llm.invoke(f"任务: {state['input']}\n生成 JSON 步骤列表")
    return {"plan": parse_steps(plan)}

def executor_node(state):
    step = state["plan"][0]
    result = execute_step(step)
    return {"past_steps": [(step, result)]}

def should_replan(state):
    if state["plan"]:
        return "executor"
    return END

workflow = StateGraph(PlanExecuteState)
workflow.add_node("planner", planner_node)
workflow.add_node("executor", executor_node)
workflow.set_entry_point("planner")
workflow.add_conditional_edges("planner", should_replan, {"executor": "executor", END: END})
workflow.add_edge("executor", "planner")

实验 4.2：DSPy 自动 Prompt 优化

目标：用 DSPy 自动搜索最优 prompt 和示例，避免手工调参。

dspy_optimize.py 
import dspy

class QASignature(dspy.Signature):
    """根据上下文回答问题"""
    context = dspy.InputField()
    question = dspy.InputField()
    answer = dspy.OutputField()

class RAG(dspy.Module):
    def __init__(self):
        super().__init__()
        self.retrieve = dspy.Retrieve(k=3)
        self.generate = dspy.ChainOfThought(QASignature)
    def forward(self, question):
        context = self.retrieve(question).passages
        return self.generate(context=context, question=question)

# 用 BootstrapFewShot 自动找最优示例

实战心法

动手的正确姿势

这不仅是教程,而是一套方法论。

🧩 1. 先跑通，再理解

拿到代码先完整跑起来。不需要完全理解每一行。先看到效果，建立信心。

🔬 2. 拆参数观测

把模型换成更弱的、温度调高、prompt 改坏 —— 观察输出变化。实验是最好的老师。

📝 3. 记笔记

每个实验后写一段复盘：这个项目解决什么问题、设计亮点是什么、我能改进哪里。

🛠 4. 改它、扩展它

每个实验都至少改动一处：加新工具、换不同场景、改成不同领域。把代码变成自己的。

🧪 5. 加测试

测试不是为了找 bug。是为了让代码思维可固化。一旦有测试，重构就不再可怕。

🚀 6. 发布它

GitHub 公开仓库、写 README、加 demo。发布这一步是和自己学习的承诺。
